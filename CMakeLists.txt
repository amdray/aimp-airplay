cmake_minimum_required(VERSION 3.20)
project(aimp_airplay VERSION 0.2.42 LANGUAGES C CXX RC)
add_compile_definitions(AIRPLAY_VERSION="${PROJECT_VERSION}")
if(NOT WIN32)
  message(FATAL_ERROR "AIMP plugins require Windows")
endif()
set(AIMP_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/sdk" CACHE PATH "Extracted AIMP 5.40 SDK")
include(cmake/raop.cmake)
set(AIRPLAY_PAIRING_SOURCES src/pairing.cpp src/pairing_transport.cpp src/pairing_plist.cpp src/pairing_credentials.cpp)
set(AIRPLAY_REMOTE_SOURCES src/remote_control.cpp src/dacp_server.cpp src/dacp_publisher.cpp src/dacp_dns.cpp)
set(AIRPLAY_ENGINE_SOURCES
  src/engine.cpp src/engine_diagnostics.cpp src/raop_session.cpp src/remote_diagnostics.cpp src/dsp_bridge.cpp
  src/crypto_runtime.cpp src/pcm.cpp src/transient_pairing.cpp)
add_library(aimp_airplay SHARED
  src/plugin.cpp src/album_art.cpp src/player_metadata.cpp
  src/airplay_menu.cpp src/receiver_controller.cpp
  src/pairing_controller.cpp src/remote_commands.cpp src/player_events.cpp
  src/receivers.cpp src/discovery.cpp ${AIRPLAY_PAIRING_SOURCES} ${AIRPLAY_REMOTE_SOURCES}
  src/pin.rc src/exports.def ${AIRPLAY_ENGINE_SOURCES})
target_compile_features(aimp_airplay PRIVATE cxx_std_17)
target_include_directories(aimp_airplay PRIVATE "${AIMP_SDK_DIR}/Sources/Cpp")
target_compile_definitions(aimp_airplay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(aimp_airplay PRIVATE user32 advapi32 bcrypt uuid mdns raop soxr)
target_include_directories(aimp_airplay PRIVATE "${RAOP_ROOT}/libcodecs/soxr/src")
target_include_directories(aimp_airplay PRIVATE "${RAOP_ROOT}/libopenssl/targets/win32/x86/include")
add_library(dsp_airplay SHARED src/dsp.cpp)
target_compile_features(dsp_airplay PRIVATE cxx_std_17)
target_link_libraries(dsp_airplay PRIVATE user32)
set_target_properties(dsp_airplay PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(dsp_airplay PRIVATE -static)
endif()
set_target_properties(aimp_airplay PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(aimp_airplay PRIVATE /W4 /utf-8)
else()
  target_compile_options(aimp_airplay PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(aimp_airplay PRIVATE -static)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_link_options(aimp_airplay PRIVATE -Wl,--kill-at)
  endif()
endif()
set(OPENSSL_RUNTIME_DIR "C:/Program Files/Git/mingw64/bin" CACHE PATH "OpenSSL 3 x64 runtime directory")
foreach(runtime IN ITEMS libcrypto-3-x64.dll libssl-3-x64.dll)
  configure_file("${OPENSSL_RUNTIME_DIR}/${runtime}" "${CMAKE_CURRENT_BINARY_DIR}/${runtime}" COPYONLY)
endforeach()
